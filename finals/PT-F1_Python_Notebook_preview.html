Chunking and Vector Index

Each PDF page is normalized, split into overlapping character chunks, and saved with a chunk ID and source-page reference. The multilingual E5 model converts each chunk into a vector. FAISS then searches those vectors to identify the most relevant evidence for a question.

In [8]:
def create_chunks(pages, chunk_size, chunk_overlap):
    chunks = []
    step = chunk_size - chunk_overlap
    for page in pages:
        text = page["text"]
        for start in range(0, len(text), step):
            piece = text[start:start + chunk_size].strip()
            if piece:
                chunks.append({"text": piece, "page": page["page"]})
    return chunks

embedding_model = SentenceTransformer(EMBEDDING_MODEL_NAME)
vectors = embedding_model.encode([item["text"] for item in chunks], normalize_embeddings=True)
index = faiss.IndexFlatIP(vectors.shape[1])
index.add(np.asarray(vectors, dtype="float32"))
